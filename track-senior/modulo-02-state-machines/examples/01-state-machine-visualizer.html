Definición de la máquina (simplificada)

const machine = {
  initial: 'idle',
  states: {
    idle:    { on: { LOAD: 'loading' } },
    loading: { on: { READY: 'ready', FAIL: 'idle' } },
    ready:   { on: { PLAY: 'playing' } },
    playing: { on: { PAUSE: 'paused', END: 'ended' } },
    paused:  { on: { PLAY: 'playing', STOP: 'idle' } },
    ended:   { on: { RESTART: 'playing', RESET: 'idle' }, type: 'final' }
  }
};
