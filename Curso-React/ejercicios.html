Ejercicio: Contador con límite — no puede pasar de 10 ni bajar de 0.

Ver solución 
function ContadorLimitado() {
  const [n, setN] = useState(0);
  return (
    <>
      <p>{n}</p>
      <button onClick={() => setN(x => Math.min(10, x + 1))} disabled={n >= 10}>+1</button>
      <button onClick={() => setN(x => Math.max(0, x - 1))} disabled={n <= 0}>-1</button>
    </>
  );
}
// Doble protección: Math.min/max EN el setter + disabled en el botón.

Ejercicio: Registro con confirm password — error si no coinciden.

Ver solución 
function Registro() {
  const [pwd, setPwd] = useState('');
  const [confirm, setConfirm] = useState('');

  const coinciden = pwd === confirm;
  const valido = pwd.length >= 8 && coinciden;

  return (
    <form onSubmit={(e) => { e.preventDefault(); alert('OK'); }}>
      <input type="password" value={pwd} onChange={(e) => setPwd(e.target.value)} placeholder="Password" />
      <input type="password" value={confirm} onChange={(e) => setConfirm(e.target.value)} placeholder="Confirmar" />
      {confirm && !coinciden && <p style={{ color: 'red' }}>No coinciden</p>}
      <button disabled={!valido}>Registrarse</button>
    </form>
  );
}

M05 Listas y Keys

Ejercicio: Paginación — mostrar 5 items por página con prev/next.

Ver solución 
function Paginada({ items }) {
  const [page, setPage] = useState(0);
  const porPagina = 5;
  const totalPaginas = Math.ceil(items.length / porPagina);
  const visibles = items.slice(page * porPagina, (page + 1) * porPagina);

  return (
    <>
      <ul>{visibles.map(i => <li key={i.id}>{i.nombre}</li>)}</ul>
      <button onClick={() => setPage(p => p - 1)} disabled={page === 0}>Anterior</button>
      <span> {page + 1} / {totalPaginas} </span>
      <button onClick={() => setPage(p => p + 1)} disabled={page >= totalPaginas - 1}>Siguiente</button>
    </>
  );
}

Ejercicio: Sort por nombre/edad sin mutar el array original.

Ver solución 
function ListaOrdenable({ usuarios }) {
  const [criterio, setCriterio] = useState('nombre');

  // ✅ Copia con spread antes de sort (sort muta)
  const ordenados = [...usuarios].sort((a, b) => {
    if (criterio === 'edad') return a.edad - b.edad;
    return a.nombre.localeCompare(b.nombre);
  });

  return (
    <>
      <button onClick={() => setCriterio('nombre')}>Por nombre</button>
      <button onClick={() => setCriterio('edad')}>Por edad</button>
      <ul>{ordenados.map(u => <li key={u.id}>{u.nombre} ({u.edad})</li>)}</ul>
    </>
  );
}

M06 useEffect

Ejercicio: Mostrar el ancho del viewport en tiempo real (con cleanup).

Ver solución 
import { useState, useEffect } from 'react';

function AnchoViewport() {
  const [width, setWidth] = useState(window.innerWidth);

  useEffect(() => {
    const handler = () => setWidth(window.innerWidth);
    window.addEventListener('resize', handler);
    return () => window.removeEventListener('resize', handler); // ⭐ cleanup
  }, []);

  return <p>Ancho: {width}px</p>;
}

Ejercicio: Fetch de posts al montar con loading/error.

Ver solución 
function Posts() {
  const [posts, setPosts] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let cancelled = false;
    fetch('https://jsonplaceholder.typicode.com/posts')
      .then(r => r.json())
      .then(data => { if (!cancelled) { setPosts(data); setLoading(false); } });
    return () => { cancelled = true; }; // anti race-condition
  }, []);

  if (loading) return <p>Cargando...</p>;
  return <ul>{posts.slice(0, 5).map(p => <li key={p.id}>{p.title}</li>)}</ul>;
}

M07-08 Hooks Avanzados y Custom Hooks

Ejercicio: Counter con useReducer (INC, DEC, RESET).

Ver solución 
import { useReducer } from 'react';

function reducer(state, action) {
  switch (action.type) {
    case 'INC': return { count: state.count + 1 };
    case 'DEC': return { count: state.count - 1 };
    case 'RESET': return { count: 0 };
    default: throw new Error('Acción desconocida');
  }
}

function Counter() {
  const [state, dispatch] = useReducer(reducer, { count: 0 });
  return (
    <>
      <p>{state.count}</p>
      <button onClick={() => dispatch({ type: 'INC' })}>+</button>
      <button onClick={() => dispatch({ type: 'DEC' })}>-</button>
      <button onClick={() => dispatch({ type: 'RESET' })}>Reset</button>
    </>
  );
}

Ejercicio: Custom hook useToggle(initial).

Ver solución 
import { useState, useCallback } from 'react';

function useToggle(initial = false) {
  const [value, setValue] = useState(initial);
  const toggle = useCallback(() => setValue(v => !v), []);
  return [value, toggle];
}

// Uso:
function Modal() {
  const [abierto, toggle] = useToggle();
  return (
    <>
      <button onClick={toggle}>Toggle</button>
      {abierto && <div>Modal</div>}
    </>
  );
}

Ejercicio: Custom hook useDebounce(value, delay).

Ver solución 
import { useState, useEffect } from 'react';

function useDebounce(value, delay = 500) {
  const [debounced, setDebounced] = useState(value);
  useEffect(() => {
    const t = setTimeout(() => setDebounced(value), delay);
    return () => clearTimeout(t); // cancelar timer anterior
  }, [value, delay]);
  return debounced;
}

// Uso: buscador que no llama API en cada tecla
function Buscador() {
  const [input, setInput] = useState('');
  const q = useDebounce(input, 400);
  useEffect(() => { if (q) console.log('Buscar:', q); }, [q]);
  return <input value={input} onChange={(e) => setInput(e.target.value)} />;
}

import { create } from 'zustand';

const useCart = create((set, get) => ({
  items: [],
  add: (p) => set(s => ({ items: [...s.items, p] })),
  remove: (id) => set(s => ({ items: s.items.filter(i => i.id !== id) })),

}));
