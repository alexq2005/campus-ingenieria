Clases JavaScript — herencia + campos privados

Abrí la consola para ver los logs. Abajo se muestran los resultados principales.

class Animal {
  #energia = 100;                        // campo PRIVADO (# es parte del nombre)
  static contador = 0;                   // estático

  constructor(nombre) {
    this.nombre = nombre;
    Animal.contador++;
  }

  get energia() { return this.#energia; }

  comer(cantidad) {
    this.#energia = Math.min(100, this.#energia + cantidad);
    return this;                         // method chaining
  }

  correr(km) {
    this.#energia = Math.max(0, this.#energia - km * 5);
    return this;
  }

  toString() { return `${this.nombre} (${this.#energia}%)`; }
}

class Perro extends Animal {
  constructor(nombre, raza) {
    super(nombre);
    this.raza = raza;
  }

  ladrar() { return `${this.nombre}: guau!`; }
  toString() { return `${super.toString()} · ${this.raza}`; }
}
